JUDUL: Estimasi Biaya Pengecatan Tembok

KAMUS:
panjang : real
 tinggi : real
  hargaCat : real
  luas : real
 totalBiaya : real

  ALGORITMA:
   MULAI

INPUT panjang
  INPUT tinggi
  INPUT hargaCat

 luas <- panjang * tinggi

 totalBiaya <- luas * hargaCat

OUTPUT "Luas Tembok: ", luas, " m2"
 OUTPUT "Total Biaya: Rp ", totalBiaya

 SELESAI
